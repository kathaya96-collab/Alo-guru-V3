Buka Google Form Anda (Mode Edit) → Klik menu titik tiga → Apps Script → Salin dan tempel seluruh kode di bawah, lalu jalankan fungsi exportALOJson() dan tempel hasilnya pada kotak di bawahnya.

Kode Google Apps Script (exportALOJson()): 
function exportALOJson() {
  var form = FormApp.getActiveForm();
  var items = form.getItems();
  var result = [];
  items.forEach(function(item) {
    var t = item.getType();
    if (t == FormApp.ItemType.MULTIPLE_CHOICE) {
      var mc = item.asMultipleChoiceItem();
      var choices = mc.getChoices();
      var opts = [], ans = 0;
      choices.forEach(function(c, i) {
        opts.push(c.getValue());
        if (c.isCorrectAnswer()) ans = i;
      });
      result.push({
        type: "PG",
        points: mc.getPoints() || 1,
        text: item.getTitle(),
        options: opts,
        answer: ans
      });
    } else if (t == FormApp.ItemType.CHECKBOX) {
      var cb = item.asCheckboxItem();
      var choicesCb = cb.getChoices();
      var optsCb = [], ansCb = [];
      choicesCb.forEach(function(c, i) {
        optsCb.push(c.getValue());
        if (c.isCorrectAnswer()) ansCb.push(i);
      });
      result.push({
        type: "PGK",
        points: cb.getPoints() || 2,
        text: item.getTitle(),
        options: optsCb,
        answer: ansCb
      });
    } else if (t == FormApp.ItemType.GRID) {
      var gr = item.asGridItem();
      var rows = gr.getRows();
      var stmts = rows.map(function(r) { return { text: r, answer: true }; });
      result.push({
        type: "BS",
        points: 2,
        text: item.getTitle(),
        statements: stmts
      });
    } else if (t == FormApp.ItemType.TEXT) {
      var tx = item.asTextItem();
      result.push({
        type: "ISIAN",
        points: tx.getPoints() || 2,
        text: item.getTitle(),
        answer: "",
        keywords: ""
      });
    }
  });
  Logger.log(JSON.stringify(result));
}
